// ============================================
// 67. Create your own Promise
// ============================================
/**
 * The core ideas:
 * 1. A promise is a state machine: pending → fulfilled, or pending → rejected. The state changes exactly once.
 * 2. then() may be called before the state settles, so callbacks have to be stored and run once it does.
 * 3. then() must run asynchronously even if the result is already there — queueMicrotask stands in for the microtask queue.
 */
class MyPromise {
  constructor(executor) {
    this.state = 'pending'; // pending / fulfilled / rejected
    this.value = undefined;
    this.callbacks = []; // then callbacks that could not run yet

    const resolve = (value) => {
      // support resolve(anotherPromise): wait for it, then take its outcome
      if (value instanceof MyPromise) {
        value.then(resolve, reject);
        return;
      }
      if (this.state !== 'pending') return; // locked — the state only changes once
      this.state = 'fulfilled';
      this.value = value;
      this.callbacks.forEach(cb => cb.onFulfilled(value));
    };

    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.callbacks.forEach(cb => cb.onRejected(reason));
    };

    try {
      executor(resolve, reject);
    } catch (err) {
      reject(err); // a synchronous throw in the executor must reject too
    }
  }

  then(onFulfilled, onRejected) {
    // when an argument is not a function, pass the value or error straight through
    onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : v => v;
    onRejected = typeof onRejected === 'function' ? onRejected : err => { throw err; };

    return new MyPromise((resolve, reject) => {
      const handleFulfilled = (value) => {
        queueMicrotask(() => { // force async, so the ordering is predictable
          try {
            resolve(onFulfilled(value)); // chaining: the return value feeds the next then
          } catch (err) {
            reject(err);
          }
        });
      };

      const handleRejected = (reason) => {
        queueMicrotask(() => {
          try {
            resolve(onRejected(reason)); // note: a caught error that is not rethrown puts the chain back on the success path
          } catch (err) {
            reject(err);
          }
        });
      };

      if (this.state === 'fulfilled') handleFulfilled(this.value);
      else if (this.state === 'rejected') handleRejected(this.value);
      else this.callbacks.push({ onFulfilled: handleFulfilled, onRejected: handleRejected }); // not settled yet — store it
    });
  }

  catch(onRejected) {
    return this.then(null, onRejected);
  }
}


// ============================================
// 123. Implement Promise.prototype.finally()
// ============================================
/**
 * The tricky part: the finally callback takes no arguments, and the result must pass through
 * unchanged whether it succeeded or failed. finally cannot alter the value or the reason —
 * it just runs something along the way.
 */
MyPromise.prototype.finally = function (callback) {
  return this.then(
    (value) => {
      callback(); // on success: run the callback, then pass the original value on
      return value;
    },
    (reason) => {
      callback(); // on failure: same, then rethrow the original reason
      throw reason;
    }
  );
};


// ============================================
// 32. Implement Promise.all()
// ============================================
/**
 * The core ideas:
 * - It only succeeds if all succeed, and the results come back in INPUT order, not completion order.
 * - The first failure rejects the whole thing immediately.
 */
function myPromiseAll(promises) {
  return new MyPromise((resolve, reject) => {
    const results = [];
    let completedCount = 0;

    if (promises.length === 0) {
      resolve([]);
      return;
    }

    promises.forEach((p, index) => {
      // wrap with Promise.resolve so plain values work the same way as promises
      MyPromise.resolve(p).then(
        (value) => {
          results[index] = value; // store by index, keeping the input order
          completedCount++;
          if (completedCount === promises.length) {
            resolve(results);
          }
        },
        (err) => reject(err) // any failure fails the whole thing at once
      );
    });
  });
}


// ============================================
// 33. Implement Promise.allSettled()
// ============================================
/**
 * How it differs from all: it waits for every promise regardless of outcome and reports
 * each one's status in detail. It never finishes early because one failed — that is the
 * biggest difference from all.
 */
function myPromiseAllSettled(promises) {
  return new MyPromise((resolve) => {
    const results = [];
    let completedCount = 0;

    if (promises.length === 0) {
      resolve([]);
      return;
    }

    promises.forEach((p, index) => {
      MyPromise.resolve(p).then(
        (value) => {
          results[index] = { status: 'fulfilled', value };
          completedCount++;
          if (completedCount === promises.length) resolve(results);
        },
        (reason) => {
          results[index] = { status: 'rejected', reason }; // record failures too; do not reject the whole
          completedCount++;
          if (completedCount === promises.length) resolve(results);
        }
      );
    });
  });
}

// 35. Implement Promise.race()
// ============================================
/**
 * The simplest one: whoever settles first wins, success or failure.
 */
function myPromiseRace(promises) {
  return new MyPromise((resolve, reject) => {
    promises.forEach((p) => {
      MyPromise.resolve(p).then(resolve, reject); // pass resolve/reject straight through; first call wins
    });
  });
}


// ============================================
// 34. Implement Promise.any()
// ============================================
/**
 * The mirror image of all: one success is enough for the whole thing to succeed.
 * It only rejects when every one failed, and it collects all the reasons (AggregateError).
 */
function myPromiseAny(promises) {
  return new MyPromise((resolve, reject) => {
    const errors = [];
    let rejectedCount = 0;

    if (promises.length === 0) {
      reject(new Error('All promises were rejected'));
      return;
    }

    promises.forEach((p, index) => {
      MyPromise.resolve(p).then(
        (value) => resolve(value), // one success settles the whole thing
        (err) => {
          errors[index] = err;
          rejectedCount++;
          if (rejectedCount === promises.length) {
            // only reject once everything failed; the native API throws AggregateError,
            // simplified here to a plain Error plus the errors array
            reject(new Error('All promises were rejected: ' + JSON.stringify(errors)));
          }
        }
      );
    });
  });
}


// MyPromise.resolve, used by the implementations above
MyPromise.resolve = function (value) {
  if (value instanceof MyPromise) return value; // already a promise — no need to wrap it again
  return new MyPromise((resolve) => resolve(value));
};
MyPromise.reject = function (reason) {
  return new MyPromise((resolve, reject) => reject(reason));
};


// ============================================
// 64. Auto-retry on rejection
// ============================================
/**
 * The situation: call an async function that may fail, retry on failure, and only report a
 * real error once the retries run out.
 * The idea: recursion — on failure, call yourself again until it succeeds or the count runs out.
 */
function retry(fn, times) {
  return new Promise((resolve, reject) => {
    function attempt(remainingTimes) {
      fn()
        .then(resolve) // success — resolve and stop
        .catch((err) => {
          if (remainingTimes <= 0) {
            reject(err); // out of attempts — really failed; surface the last error
          } else {
            attempt(remainingTimes - 1); // attempts left — try again
          }
        });
    }
    attempt(times);
  });
}

/**
 * Example:
 *
 * let count = 0;
 * function mayFail() {
 *   return new Promise((resolve, reject) => {
 *     count++;
 *     if (count < 3) reject('failed attempt ' + count);
 *     else resolve('success on attempt ' + count);
 *   });
 * }
 *
 * retry(mayFail, 5).then(console.log); // fails twice, succeeds on the third: "success on attempt 3"

// ============================================
// 92. Throttle promises (at most N at a time)
// ============================================
/**
 * The situation: you have 100 async tasks — 100 network requests, say — and you do not want to
 * fire all of them at once and flatten the server. Run at most `max` at a time, and start a new
 * one each time one finishes, until they are all done.
 *
 * The idea:
 * 1. Keep a counter of how many are currently running.
 * 2. Whenever one finishes, check the queue and immediately start the next.
 */
class PromisePool {
  constructor(max) {
    this.max = max; // how many may run at once
    this.running = 0; // how many are running now
    this.queue = []; // tasks waiting their turn
  }

  add(promiseCreator) {
    // promiseCreator is a FUNCTION that returns a promise, not a promise.
    // This matters: a promise starts executing the moment it is created, and we need to defer
    // the start until this task's turn comes. Hence a function, not an already-running promise.
    return new Promise((resolve, reject) => {
      const task = () => {
        this.running++;
        promiseCreator()
          .then((result) => {
            resolve(result);
            this._next();
          })
          .catch((err) => {
            reject(err);
            this._next();
          });
      };

      if (this.running < this.max) {
        task(); // under the limit — run now
      } else {
        this.queue.push(task); // at the limit — queue it
      }
    });
  }

  _next() {
    this.running--;
    if (this.queue.length > 0) {
      const nextTask = this.queue.shift();
      nextTask(); // something is waiting — start it straight away
    }
  }
}

/**
 * Example: at most two at a time
 *
 * const pool = new PromisePool(2);
 * function createTask(id, delay) {
 *   return () => new Promise(resolve => {
 *     console.log('start task', id);
 *     setTimeout(() => {
 *       console.log('finish task', id);
 *       resolve(id);
 *     }, delay);
 *   });
 * }
 *
 * pool.add(createTask(1, 1000));
 * pool.add(createTask(2, 1000));
 * pool.add(createTask(3, 1000)); // waits for 1 or 2 to finish before starting
 * pool.add(createTask(4, 1000));
 */


module.exports = {
  MyPromise,
  myPromiseAll,
  myPromiseAllSettled,
  myPromiseRace,
  myPromiseAny,
  retry,
  PromisePool,
};
