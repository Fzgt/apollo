// 230. 二叉搜索树中的第K小元素
function kthSmallest(root: TreeNode | null, k: number): number {
	const sortedArr: number[] = [];

	const inorder = (node: TreeNode | null) => {
		if (!node) return;
		inorder(node.left);
		sortedArr.push(node.val);
		inorder(node.right);
	};

	inorder(root);
	return sortedArr[k - 1];
}
